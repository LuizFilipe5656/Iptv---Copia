'use client'

import { useEffect, useState } from 'react'
import { useAuth } from '@/contexts/AuthContext'
import { supabase } from '@/lib/supabase'
import { ResellerRoute } from '@/components/layout/ResellerRoute'
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle, DialogTrigger } from '@/components/ui/dialog'
import { Label } from '@/components/ui/label'
import { Users, Wallet, Plus, Calendar, Ban, ShieldCheck } from 'lucide-react'
import type { User } from '@/types/user'

export default function ResellerPage() {
  const { user } = useAuth()
  const [clients, setClients] = useState<User[]>([])
  const [loading, setLoading] = useState(true)
  const [modalOpen, setModalOpen] = useState(false)
  const [createData, setCreateData] = useState({
    name: '',
    email: '',
    password: '',
    plan: 'FREE',
    days: '30',
  })
  const [error, setError] = useState('')

  useEffect(() => {
    if (user) {
      fetchClients()
    }
  }, [user])

  const fetchClients = async () => {
    try {
      const { data, error } = await supabase
        .from('users')
        .select('*')
        .eq('created_by', user?.id)
        .order('created_at', { ascending: false })

      if (error) throw error
      setClients(data || [])
    } catch (error) {
      console.error('Error fetching clients:', error)
    } finally {
      setLoading(false)
    }
  }

  const createClient = async () => {
    setError('')

    if (user?.credits! <= 0) {
      setError('Créditos insuficientes')
      return
    }

    if (!createData.name || !createData.email || !createData.password) {
      setError('Preencha todos os campos')
      return
    }

    try {
      // Create auth user
      const { data: authData, error: authError } = await supabase.auth.signUp({
        email: createData.email,
        password: createData.password,
      })

      if (authError) throw authError

      if (authData.user) {
        // Create user profile
        const expirationDate = new Date(Date.now() + parseInt(createData.days) * 24 * 60 * 60 * 1000).toISOString()

        const { error: profileError } = await supabase.from('users').insert({
          id: authData.user.id,
          email: createData.email,
          name: createData.name,
          plan: createData.plan,
          expiration_date: expirationDate,
          status: 'ACTIVE',
          role: 'user',
          created_by: user?.id,
          credits: 0,
        })

        if (profileError) throw profileError

        // Consume 1 credit from reseller
        const newCredits = (user?.credits || 0) - 1
        const { error: creditError } = await supabase
          .from('users')
          .update({ credits: newCredits })
          .eq('id', user?.id)

        if (creditError) throw creditError

        // Update local user state
        if (user) {
          user.credits = newCredits
        }

        setModalOpen(false)
        setCreateData({ name: '', email: '', password: '', plan: 'FREE', days: '30' })
        await fetchClients()
      }
    } catch (err: any) {
      setError(err.message || 'Erro ao criar cliente')
    }
  }

  const addDays = async (clientId: string) => {
    try {
      const client = clients.find((c) => c.id === clientId)
      if (!client) return

      const newExpiration = new Date(
        new Date(client.expiration_date).getTime() + 30 * 24 * 60 * 60 * 1000
      ).toISOString()

      const { error } = await supabase
        .from('users')
        .update({ expiration_date: newExpiration })
        .eq('id', clientId)

      if (error) throw error
      await fetchClients()
    } catch (error) {
      console.error('Error adding days:', error)
    }
  }

  const banClient = async (clientId: string) => {
    try {
      const { error } = await supabase
        .from('users')
        .update({ status: 'BANNED' })
        .eq('id', clientId)

      if (error) throw error
      await fetchClients()
    } catch (error) {
      console.error('Error banning client:', error)
    }
  }

  const unbanClient = async (clientId: string) => {
    try {
      const { error } = await supabase
        .from('users')
        .update({ status: 'ACTIVE' })
        .eq('id', clientId)

      if (error) throw error
      await fetchClients()
    } catch (error) {
      console.error('Error unbanning client:', error)
    }
  }

  if (loading) {
    return (
      <ResellerRoute>
        <div className="min-h-screen flex items-center justify-center bg-background">
          <div className="text-foreground">Carregando...</div>
        </div>
      </ResellerRoute>
    )
  }

  return (
    <ResellerRoute>
      <div className="min-h-screen bg-background p-8">
        <div className="max-w-7xl mx-auto">
          <div className="mb-8">
            <h1 className="text-3xl font-bold text-foreground">Painel Revendedor</h1>
            <p className="text-muted-foreground mt-2">Gerencie seus clientes</p>
          </div>

          {/* Credits Card */}
          <Card className="border-border bg-card mb-8">
            <CardHeader className="flex flex-row items-center justify-between">
              <div className="flex items-center gap-4">
                <Wallet className="h-8 w-8 text-primary" />
                <div>
                  <CardTitle className="text-2xl">{user?.credits || 0} Créditos Disponíveis</CardTitle>
                </div>
              </div>
              <Dialog open={modalOpen} onOpenChange={setModalOpen}>
                <DialogTrigger asChild>
                  <Button>
                    <Plus className="w-4 h-4 mr-2" />
                    Criar Cliente
                  </Button>
                </DialogTrigger>
                <DialogContent>
                  <DialogHeader>
                    <DialogTitle>Criar Novo Cliente</DialogTitle>
                    <DialogDescription>
                      Preencha os dados para criar um novo cliente. Será consumido 1 crédito.
                    </DialogDescription>
                  </DialogHeader>
                  <div className="space-y-4 py-4">
                    <div className="space-y-2">
                      <Label htmlFor="name">Nome</Label>
                      <Input
                        id="name"
                        value={createData.name}
                        onChange={(e) => setCreateData({ ...createData, name: e.target.value })}
                        placeholder="Nome do cliente"
                      />
                    </div>
                    <div className="space-y-2">
                      <Label htmlFor="email">Email</Label>
                      <Input
                        id="email"
                        type="email"
                        value={createData.email}
                        onChange={(e) => setCreateData({ ...createData, email: e.target.value })}
                        placeholder="cliente@email.com"
                      />
                    </div>
                    <div className="space-y-2">
                      <Label htmlFor="password">Senha</Label>
                      <Input
                        id="password"
                        type="password"
                        value={createData.password}
                        onChange={(e) => setCreateData({ ...createData, password: e.target.value })}
                        placeholder="••••••••"
                      />
                    </div>
                    <div className="space-y-2">
                      <Label htmlFor="plan">Plano</Label>
                      <Input
                        id="plan"
                        value={createData.plan}
                        onChange={(e) => setCreateData({ ...createData, plan: e.target.value })}
                        placeholder="FREE"
                      />
                    </div>
                    <div className="space-y-2">
                      <Label htmlFor="days">Dias de Expiração</Label>
                      <Input
                        id="days"
                        type="number"
                        value={createData.days}
                        onChange={(e) => setCreateData({ ...createData, days: e.target.value })}
                        placeholder="30"
                      />
                    </div>
                    {error && (
                      <div className="text-sm text-destructive">{error}</div>
                    )}
                  </div>
                  <DialogFooter>
                    <Button variant="outline" onClick={() => setModalOpen(false)}>
                      Cancelar
                    </Button>
                    <Button onClick={createClient} disabled={user?.credits! <= 0}>
                      Criar Cliente
                    </Button>
                  </DialogFooter>
                </DialogContent>
              </Dialog>
            </CardHeader>
          </Card>

          {/* Stats */}
          <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-8">
            <Card className="border-border bg-card">
              <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
                <CardTitle className="text-sm font-medium text-muted-foreground">Total Clientes</CardTitle>
                <Users className="h-4 w-4 text-muted-foreground" />
              </CardHeader>
              <CardContent>
                <div className="text-2xl font-bold text-foreground">{clients.length}</div>
              </CardContent>
            </Card>

            <Card className="border-border bg-card">
              <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-2">
                <CardTitle className="text-sm font-medium text-muted-foreground">Clientes Ativos</CardTitle>
                <ShieldCheck className="h-4 w-4 text-muted-foreground" />
              </CardHeader>
              <CardContent>
                <div className="text-2xl font-bold text-foreground">
                  {clients.filter((c) => c.status === 'ACTIVE').length}
                </div>
              </CardContent>
            </Card>
          </div>

          {/* Clients Table */}
          <Card className="border-border bg-card">
            <CardHeader>
              <CardTitle>Meus Clientes</CardTitle>
            </CardHeader>
            <CardContent className="p-0">
              <div className="overflow-x-auto">
                <table className="w-full">
                  <thead>
                    <tr className="border-b border-border">
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Nome</th>
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Email</th>
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Plano</th>
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Expiração</th>
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Status</th>
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Data Criação</th>
                      <th className="text-left p-4 text-sm font-medium text-muted-foreground">Ações</th>
                    </tr>
                  </thead>
                  <tbody>
                    {clients.map((client) => (
                      <tr key={client.id} className="border-b border-border hover:bg-muted/50">
                        <td className="p-4 text-sm text-foreground">{client.name || '-'}</td>
                        <td className="p-4 text-sm text-foreground">{client.email}</td>
                        <td className="p-4 text-sm text-muted-foreground">{client.plan}</td>
                        <td className="p-4 text-sm text-muted-foreground">
                          {new Date(client.expiration_date).toLocaleDateString('pt-BR')}
                        </td>
                        <td className="p-4">
                          <span
                            className={`inline-flex items-center px-2 py-1 rounded-full text-xs font-medium ${
                              client.status === 'ACTIVE'
                                ? 'bg-green-500/20 text-green-500 border border-green-500/30'
                                : 'bg-red-500/20 text-red-500 border border-red-500/30'
                            }`}
                          >
                            {client.status}
                          </span>
                        </td>
                        <td className="p-4 text-sm text-muted-foreground">
                          {new Date(client.created_at).toLocaleDateString('pt-BR')}
                        </td>
                        <td className="p-4">
                          <div className="flex gap-2">
                            <Button
                              size="sm"
                              variant="outline"
                              onClick={() => addDays(client.id)}
                            >
                              <Calendar className="w-4 h-4" />
                            </Button>
                            {client.status === 'ACTIVE' ? (
                              <Button
                                size="sm"
                                variant="outline"
                                onClick={() => banClient(client.id)}
                              >
                                <Ban className="w-4 h-4" />
                              </Button>
                            ) : (
                              <Button
                                size="sm"
                                variant="outline"
                                onClick={() => unbanClient(client.id)}
                              >
                                <ShieldCheck className="w-4 h-4" />
                              </Button>
                            )}
                          </div>
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            </CardContent>
          </Card>
        </div>
      </div>
    </ResellerRoute>
  )
}
